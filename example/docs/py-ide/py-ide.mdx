# Py IDE

The `<py-ide>` component is an in-browser Python IDE powered by Pyodide. It provides:

- **Multi-file editing** with tabs
- **[CodeMirror](https://codemirror.net)** editor with Python syntax highlighting
- **Run/Save/Clear** controls and resizable panes
- **Live output** including inline images for <code>[matplotlib](https://matplotlib.org)</code> plots
- **Local persistence** via `localStorage`

## Import

```javascript
import "essential-components/src/py-ide";
```

## Basic Usage

```html
<py-ide></py-ide>
```

## Usage with Passing Files

```html
<py-ide>
  <py-file name="main.py">print("Hello from PyIDE 👋")</py-file>
  <py-file name="utils.py">def sqr(x): return x*x</py-file>
</py-ide>

<!-- or, you can use <script> tags -->

<py-ide>
  <script type="text/plain" data-filename="main.py">
    print("Hello from PyIDE 👋")
    for i in range(3):
        print(f"Line {i+1}")
  </script>
  <script type="text/plain" data-filename="utils.py">
    def sqr(x): return x*x
  </script>
</py-ide>
```

## Local Storage (storage-key)

- **what**: A unique key used to persist the IDE state in `localStorage` (files, open tabs, active file, and last output)
- **default**: `"py-ide"`
- **why**: Use different keys to prevent pages/widgets from overwriting each other

### Set a custom key (HTML)

```html
<py-ide storage-key="docs-pyide-intro">
  <script type="text/plain" data-filename="main.py">
    print("Hello")
  </script>
</py-ide>
```

### Set a custom key (React/MDX)

```tsx
import { PyIDE } from "@site/src/components/py-ide";

<PyIDE storageKey="docs-pyide-intro" />;
```

### Clear saved state

```js
localStorage.removeItem("docs-pyide-intro");
```

> Tip: If you need truly isolated sessions for demos, give each demo a different `storage-key`.

## Programmatic API

Access properties and methods from JavaScript:

```js
const ide = document.querySelector("py-ide");

console.log(ide.code); // [{ name: "main.py", content: "print('Hello from PyIDE 👋')..." }, ...]
console.log(ide.output); // Hello from PyIDE 👋...

// Set files
ide.setCode([{ name: "example.py", content: 'print("hi")' }]);

// Listen to events
ide.addEventListener("input", (e) => console.log("input", e.detail));
ide.addEventListener("change", (e) => console.log("change", e.detail));
ide.addEventListener("submit", (e) => console.log("submit", e.detail));
ide.addEventListener("save", (e) => console.log("save", e.detail));
```

### Event Integration (Vanilla JS)

```html
<py-ide id="demo-ide" storage-key="docs-pyide-events">
  <script type="text/plain" data-filename="main.py">
    print("Hello from events example")
  </script>
</py-ide>

<button id="getCodeBtn">Get Code</button>
<button id="getOutputBtn">Get Output</button>
<pre id="codeDisplay"></pre>
<pre id="outputDisplay"></pre>

<script>
  const ide = document.getElementById("demo-ide");
  const codeDisplay = document.getElementById("codeDisplay");
  const outputDisplay = document.getElementById("outputDisplay");
  const getCodeBtn = document.getElementById("getCodeBtn");
  const getOutputBtn = document.getElementById("getOutputBtn");

  // Listen to events
  ide.addEventListener("input", (e) => console.log("INPUT", e.detail));
  ide.addEventListener("change", (e) => console.log("CHANGE", e.detail));
  ide.addEventListener("submit", (e) => console.log("SUBMIT", e.detail));
  ide.addEventListener("save", (e) => console.log("SAVE", e.detail));

  getCodeBtn.addEventListener("click", () => {
    codeDisplay.textContent = JSON.stringify(ide.code, null, 2);
  });
  getOutputBtn.addEventListener("click", () => {
    outputDisplay.textContent = ide.output;
  });
</script>
```

### Event Integration (React)

```tsx
import React, { useEffect, useRef } from "react";
import "essential-components/src/py-ide";

export default function PyIDEInteractiveDemo() {
  const pyIdeRef = useRef<HTMLElement | null>(null);

  useEffect(() => {
    const el = pyIdeRef.current;

    if (!el) {
      console.warn("py-ide element not found in the DOM.");
      return;
    }

    const onInput = (e: Event) => {
      const customEvent = e as CustomEvent;
      console.log("INPUT", customEvent.detail);
    };
    const onChange = (e: Event) => {
      const customEvent = e as CustomEvent;
      console.log("CHANGE", customEvent.detail);
    };
    const onSubmit = (e: Event) => {
      const customEvent = e as CustomEvent;
      console.log("SUBMIT", customEvent.detail);
    };
    const onSave = (e: Event) => {
      const customEvent = e as CustomEvent;
      console.log("SAVE", customEvent.detail);
    };

    el.addEventListener("input", onInput);
    el.addEventListener("change", onChange);
    el.addEventListener("submit", onSubmit);
    el.addEventListener("save", onSave);

    return () => {
      el.removeEventListener("input", onInput);
      el.removeEventListener("change", onChange);
      el.removeEventListener("submit", onSubmit);
      el.removeEventListener("save", onSave);
    };
  }, []);

  return (
    <py-ide ref={pyIdeRef} storage-key="docs-pyide-react">
      <script type="text/plain" data-filename="main.py">
        {"print('React events demo')"}
      </script>
    </py-ide>
  );
}
```
